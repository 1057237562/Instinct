cmake_minimum_required(VERSION 3.21)
project(instinct_cupti_trace LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
set(CUPTI_ROOT "${CUDAToolkit_LIBRARY_ROOT}/extras/CUPTI" CACHE PATH "CUPTI SDK")
find_path(CUPTI_INCLUDE cupti.h HINTS "${CUPTI_ROOT}/include" REQUIRED)
find_library(CUPTI_LIBRARY cupti HINTS "${CUPTI_ROOT}/lib64" REQUIRED)
add_library(instinct_cupti_trace SHARED trace.cpp)
target_compile_features(instinct_cupti_trace PRIVATE cxx_std_17)
target_include_directories(instinct_cupti_trace PRIVATE "${CUPTI_INCLUDE}" ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(instinct_cupti_trace PRIVATE "${CUPTI_LIBRARY}")
if(MSVC)
  target_compile_options(instinct_cupti_trace PRIVATE /EHsc /utf-8)
endif()
