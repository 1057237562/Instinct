cmake_minimum_required(VERSION 3.21)
project(instinct_cublaslt LANGUAGES CXX)
find_package(CUDAToolkit 12.8 REQUIRED)
add_library(instinct_cublaslt SHARED bridge.cpp)
target_compile_features(instinct_cublaslt PRIVATE cxx_std_17)
target_link_libraries(instinct_cublaslt PRIVATE CUDA::cublasLt)
set_target_properties(instinct_cublaslt PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(instinct_cublaslt PRIVATE /W4 /EHsc)
else()
  target_compile_options(instinct_cublaslt PRIVATE -Wall -Wextra -Werror)
endif()

option(INSTINCT_BUILD_CUDNN "Build the optional native packed attention bridge" OFF)
if(INSTINCT_BUILD_CUDNN)
  set(CUDNN_ROOT "" CACHE PATH "cuDNN SDK root")
  set(CUDNN_FRONTEND_ROOT "" CACHE PATH "cuDNN Frontend source root")
  find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS "${CUDNN_ROOT}/include" REQUIRED)
  find_path(CUDNN_FRONTEND_INCLUDE_DIR cudnn_frontend.h HINTS "${CUDNN_FRONTEND_ROOT}/include" REQUIRED)
  find_library(CUDNN_LIBRARY NAMES cudnn cudnn64_9 HINTS "${CUDNN_ROOT}/lib" "${CUDNN_ROOT}/lib/x64" REQUIRED)
  add_library(native_sdpa SHARED attention.cpp)
  target_compile_features(native_sdpa PRIVATE cxx_std_17)
  target_include_directories(native_sdpa PRIVATE "${CUDNN_INCLUDE_DIR}" "${CUDNN_FRONTEND_INCLUDE_DIR}")
  target_link_libraries(native_sdpa PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc "${CUDNN_LIBRARY}")
  if(MSVC)
    target_compile_options(native_sdpa PRIVATE /bigobj /EHsc /utf-8)
  endif()
endif()
