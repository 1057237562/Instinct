cmake_minimum_required(VERSION 3.23)
project(native_sdpa_probe LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(native_sdpa SHARED "${REPO}/model/dense_cublas/native/attention.cpp")
target_compile_features(native_sdpa PRIVATE cxx_std_17)
target_include_directories(native_sdpa PRIVATE "${REPO}/.cache/cudnn-frontend/include" "${REPO}/.cache/native-libs/nvidia/cudnn/include")
target_link_libraries(native_sdpa PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc "${REPO}/.cache/cudnn-import/cudnn.lib")
if(MSVC)
  target_compile_options(native_sdpa PRIVATE /bigobj /EHsc)
endif()
